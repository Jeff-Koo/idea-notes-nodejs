========== Learn about aggregate ==========

    - Example for explain $lookup and $unwind 
    - Set the sample data with 2 collections 

use test

db.classes.insertMany([
    { _id : "classA", data : "A" }, 
    { _id : "classB", data : "B" }, 
    { _id : "classC", data : "C" }, 
    { _id : "classD", data : "D" }, 
])

db.students.insertMany([
    { _id : "student1", classData : "A" },
    { _id : "student2", classData : "A" },
    { _id : "student3", classData : "A" },
    { _id : "student4", classData : "B" },
    { _id : "student5", classData : "B" },
    { _id : "student6", classData : "D" },
])


========== $lookup ==========

db.classes.aggregate([
    {
        '$lookup': {
            'from': 'students', 
            'localField': 'data', 
            'foreignField': 'classData', 
            'as': 'studentsInClass'
        }
    }
])


result (contains multiple Objects in studentsInClass[]):

the documents in collection 'students' which has 'classData' == 'A'
will push into studentsInClass[] of the object from 'classes' with data : 'A' 


{ _id: 'classA',
  data: 'A',
  studentsInClass: 
  [ { _id: 'student1', classData: 'A' },
    { _id: 'student2', classData: 'A' },
    { _id: 'student3', classData: 'A' } ] 
}

{ _id: 'classB',
  data: 'B',
  studentsInClass: 
  [ { _id: 'student4', classData: 'B' },
    { _id: 'student5', classData: 'B' } ] 
}

{ _id: 'classC', data: 'C', studentsInClass: []}

{ _id: 'classD',
  data: 'D',
  studentsInClass: [ { _id: 'student6', classData: 'D' } ] 
}



========== $unwind ==========

// add a stage of $unwind afte $lookup
db.classes.aggregate([
    {
        '$lookup': {
            'from': 'students', 
            'localField': 'data', 
            'foreignField': 'classData', 
            'as': 'studentsInClass'
        }
    },
    {
        '$unwind' : {
            path: '$studentsInClass',
            preserveNullAndEmptyArrays: true
        }
    }
])

result (split the array, 1 document is splited into 3 documents) : 

{ _id: 'classA',
  data: 'A',
  studentsInClass: { _id: 'student1', classData: 'A' } 
}
  
{ _id: 'classA',
  data: 'A',
  studentsInClass: { _id: 'student2', classData: 'A' } 
}

{ _id: 'classA',
  data: 'A',
  studentsInClass: { _id: 'student3', classData: 'A' } 
}

{ _id: 'classB',
  data: 'B',
  studentsInClass: { _id: 'student4', classData: 'B' } 
}

{ _id: 'classB',
  data: 'B',
  studentsInClass: { _id: 'student5', classData: 'B' }
}

{ _id: 'classC', data: 'C' }    <-- this records remains because "preserveNullAndEmptyArrays: true"

{ _id: 'classD',
  data: 'D',
  studentsInClass: { _id: 'student6', classData: 'D' } 
}
